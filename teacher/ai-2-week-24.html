Run the complete scripted comparison

def simulate(plan=None):
    exposures = {"A": 0, "B": 0}
    clicks = {"A": 0, "B": 0}
    scripted_click = {"A": True, "B": False}
    log = []
    for round_number in range(3):
        if plan is None:
            shown = sorted(clicks, key=lambda name: (-clicks[name], name))[0]
        else:
            shown = plan[round_number]
        exposures[shown] += 1
        clicked = scripted_click[shown]
        if clicked:
            clicks[shown] += 1
        log.append((round_number + 1, shown, clicked, exposures.copy(), clicks.copy()))
    return log, exposures, clicks
for label, plan in [("click-count policy", None), ("planned exposure", ["A", "B", "A"])]:
    log, exposures, clicks = simulate(plan)
    print(label)
    for event in log:
        print(event)
    print("displays", exposures, "clicks", clicks)
