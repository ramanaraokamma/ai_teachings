Block actionPython and effect

def count_short_books(page_counts):
    # Supplied input: a list of nonnegative integer page counts.
    count = 0
    for pages in page_counts:
        if pages < 100:
            count = count + 1
    return count

print(count_short_books([80, 150, 90]))
print(count_short_books([80, 150, 100]))
print(count_short_books([]))
