training = [(0, 1), (1, 3), (2, 5)]
candidates = [(1, 1), (2, 0), (2, 1)]
def sse(line):
    m, b = line
    return sum((y - (m * x + b)) ** 2 for x, y in training)
chosen = min(candidates, key=sse)
print([sse(line) for line in candidates])
print(chosen)
# [5, 3, 0], then (2, 1); first candidate wins ties.

from statistics import mean


def fit_training_line(training, candidates):
    if not training or not candidates:
        raise ValueError('Need nonempty training and declared candidate set')
    scored = []
    for m, b in candidates:
        sse = sum((y - (m * x + b)) ** 2 for x, y in training)
        scored.append(sse)
    best_index = min(range(len(candidates)), key=lambda i: scored[i])
    return candidates[best_index], scored


def evaluate(line, evaluation):
    if not evaluation:
        raise ValueError('No observed error average without evaluation records')
    m, b = line
    predictions = [m * x + b for x, y in evaluation]
    residuals = [y - p for (x, y), p in zip(evaluation, predictions)]
    return predictions, residuals, mean(abs(r) for r in residuals), mean(r*r for r in residuals)


# Caller must supply finite numeric pairs in declared day/cm units.
training = [(0, 1), (1, 3), (2, 5)]
candidates = [(1, 1), (2, 0), (2, 1)]
line, training_sse = fit_training_line(training, candidates)
validation = [(3, 6), (4, 10)]
print(training_sse, line)
print(evaluate(line, validation))
constant = mean(y for x, y in training)
print('training mean', constant, evaluate((0, constant), validation))

Read the method and complete arithmetic bridge

Estimate slope and intercept from training pairs using least squares. Ages are weeks and heights are cm. Test targets are excluded from fitting; training ages must include distinct values.

A straight line predicts y = slope × x + intercept. Slope is the change in predicted height for one extra week; intercept is the prediction at x=0, which may lie outside the measured ages.

Complete formula-based fit/test script

"""Fit a line to training pairs and evaluate a separate example."""
train = [(1, 3), (2, 5), (3, 7)]
if not train:
    raise ValueError("Training examples are required")
n = len(train)
x_mean = sum(x for x, y in train) / n
y_mean = sum(y for x, y in train) / n
num = sum((x - x_mean) * (y - y_mean) for x, y in train)
den = sum((x - x_mean) ** 2 for x, y in train)
if den == 0:
    raise ValueError("Distinct training ages are required")
slope = num / den
intercept = y_mean - slope * x_mean
test = [(4, 10)]
if not test:
    raise ValueError("Test examples are required for MAE")
preds = [slope * x + intercept for x, y in test]
mae = sum(abs(y - pred) for pred, (x, y) in zip(preds, test)) / len(test)
print(slope, intercept, preds, mae)
