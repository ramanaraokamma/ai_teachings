Separate graph, budget and evidence checks

ALLOWED = {'ready': {'looking_up'}, 'looking_up': {'review', 'done'},
           'review': {'done'}, 'done': set()}

def audit_trace(trace, budget=2):
    if type(budget) is not int or budget < 0:
        raise ValueError('budget must be a nonnegative integer')
    if not trace or trace[0] != 'ready':
        return 'invalid start'
    attempts = 0
    for before, after in zip(trace, trace[1:]):
        if after not in ALLOWED.get(before, set()):
            return f'invalid edge: {before}->{after}'
        if after == 'looking_up':
            if attempts >= budget:
                return 'lookup budget exceeded'
            attempts += 1
    return f"{'complete' if trace[-1] == 'done' else 'incomplete'}; lookups={attempts}"

for trace, budget in [(['ready','looking_up','done'], 2),
                      (['ready','looking_up','review','looking_up','done'], 2),
                      (['ready','looking_up','done'], 0),
                      (['ready','looking_up'], 2)]:
    print(audit_trace(trace, budget))

Interpret the checks and their limits

Predict the four printed lines: complete; lookups=1, invalid edge: review->looking_up, lookup budget exceeded, incomplete; lookups=1. The function checks graph edges and a lookup-entry budget only. It accepts review→done structurally but does not verify the required authorised evidence; no tool output, external call or authorisation record is simulated. “complete” means the structural trace ended at done, not that the goal is correct or an action is authorised.

def transition(state, target):
    allowed = {'ready': {'looking_up'}, 'looking_up': {'review', 'done'}, 'review': {'done'}, 'done': set()}
    if target not in allowed.get(state, set()):
        raise ValueError('invalid transition')
    return target
